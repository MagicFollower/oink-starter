    /**
     * B2: 自定义样式不被覆盖（style index != 0 时保留原样式）
     */
    @Test
    public void testFillData_CustomStyleNotOverwritten() throws Exception {
        List<String> titles = Arrays.asList("Col1");
        Map<String, List<String>> data = new LinkedHashMap<>();
        data.put("S1", Collections.singletonList("value"));

        ExcelExportUtils<String> util = ExcelExportUtils.init(titles, data);
        // 提前创建自定义样式（避免在 lambda 中抛出 checked exception）
        Workbook wbBeforeFill = getWorkbook(util);
        CellStyle customStyle = wbBeforeFill.createCellStyle();
        customStyle.setFillForegroundColor(org.apache.poi.ss.usermodel.IndexedColors.YELLOW.getIndex());
        customStyle.setFillPattern(FillPatternType.SOLID_FOREGROUND);

        util.fillData((dataMap, sheetMap) -> {
            Sheet sheet = sheetMap.get("S1");
            Row row = sheet.createRow(1);
            Cell cell = row.createCell(0);
            cell.setCellValue("value");
            // 设置自定义样式（索引应 > 0）
            cell.setCellStyle(customStyle);
        });

        Workbook wb = getWorkbook(util);
        Sheet sheet = wb.getSheetAt(0);
        Cell cell = sheet.getRow(1).getCell(0);

        // 自定义样式索引 > 0，不应被默认样式覆盖
        assertTrue("自定义样式索引应 > 0", cell.getCellStyle().getIndex() > 0);
        assertEquals("自定义样式应保留",
                org.apache.poi.ss.usermodel.IndexedColors.YELLOW.getIndex(),
                cell.getCellStyle().getFillForegroundColor());

        wb.close();
    }

            @Override

            @Override
            public void write(int b) throws java.io.IOException { bos.write(b); }
            @Override
            public void write(byte[] b, int off, int len) throws java.io.IOException { bos.write(b, off, len); }
            @Override
            public void flush() throws java.io.IOException { bos.flush(); }

        verify(response).getOutputStream();

        assertTrue("输出字节应 > 0", bos.toByteArray().length > 0);

    /**
     * E1: 从 XSSFWorkbook 复制到另一个 XSSFWorkbook，验证行数、单元格值、合并区域一致
     */
    @Test
    public void testCopySheet_ContentAndStylePreserved() throws Exception {
        // ---- 构建源 Workbook ----
        XSSFWorkbook sourceWb = new XSSFWorkbook();
        XSSFSheet sourceSheet = (XSSFSheet) sourceWb.createSheet("源Sheet");

        // 第 0 行：表头
        Row headerRow = sourceSheet.createRow(0);
        Cell hCell0 = headerRow.createCell(0);
        hCell0.setCellValue("标题A");
        CellStyle headerStyle = sourceWb.createCellStyle();
        headerStyle.setFillForegroundColor(org.apache.poi.ss.usermodel.IndexedColors.GREY_25_PERCENT.getIndex());
        headerStyle.setFillPattern(FillPatternType.SOLID_FOREGROUND);
        hCell0.setCellStyle(headerStyle);
        headerRow.createCell(1).setCellValue("标题B");

        // 第 1 行：数据
        Row dataRow = sourceSheet.createRow(1);
        dataRow.createCell(0).setCellValue("数据1");
        dataRow.createCell(1).setCellValue(100.5);

        // 第 2 行：布尔值
        Row boolRow = sourceSheet.createRow(2);
        boolRow.createCell(0).setCellValue(true);
        boolRow.createCell(1).setBlank();

        // 合并区域：A1:B1
        sourceSheet.addMergedRegion(new CellRangeAddress(0, 0, 0, 1));

        // 设置列宽
        sourceSheet.setColumnWidth(0, 5000);
        sourceSheet.setColumnWidth(1, 8000);

        // ---- 复制到目标 Workbook ----
        XSSFWorkbook targetWb = new XSSFWorkbook();
        ExcelExportUtils.copySheet(sourceSheet, targetWb, "复制Sheet");

        // ---- 验证 ----
        Sheet targetSheet = targetWb.getSheet("复制Sheet");
        assertNotNull("目标 Sheet 应存在", targetSheet);

        // 行数一致
        assertEquals(2, targetSheet.getLastRowNum());

        // 合并区域一致
        assertEquals("合并区域数量应一致", 1, targetSheet.getNumMergedRegions());

        // 列宽一致
        assertEquals(5000, targetSheet.getColumnWidth(0));
        assertEquals(8000, targetSheet.getColumnWidth(1));

        // 单元格值一致
        assertEquals("标题A", targetSheet.getRow(0).getCell(0).getStringCellValue());
        assertEquals("标题B", targetSheet.getRow(0).getCell(1).getStringCellValue());
        assertEquals("数据1", targetSheet.getRow(1).getCell(0).getStringCellValue());
        // copyCell 将 NUMERIC 转为 STRING 存储（使用 BigDecimal.plainString）
        assertEquals("100.5", targetSheet.getRow(1).getCell(1).getStringCellValue());
        assertTrue(targetSheet.getRow(2).getCell(0).getBooleanCellValue());
        assertEquals(CellType.BLANK, targetSheet.getRow(2).getCell(1).getCellType());

        // 样式保留：表头单元格应有灰色填充
        CellStyle copiedStyle = targetSheet.getRow(0).getCell(0).getCellStyle();
        assertEquals(org.apache.poi.ss.usermodel.IndexedColors.GREY_25_PERCENT.getIndex(),
                copiedStyle.getFillForegroundColor());

        sourceWb.close();
        targetWb.close();
    }

        Workbook wb = new HSSFWorkbook();
        Sheet sheet = wb.createSheet("Test");
        Row row = sheet.createRow(0);

        // STRING 类型
        Cell stringCell = row.createCell(0);
        stringCell.setCellValue("Hello");
        assertEquals("Hello", ExcelExportUtils.getStringCellValue(stringCell));

        // NUMERIC 类型（整数）
        Cell numCell = row.createCell(1);
        numCell.setCellValue(42);
        assertEquals("42", ExcelExportUtils.getStringCellValue(numCell));

        // NUMERIC 类型（小数）
        Cell decCell = row.createCell(2);
        decCell.setCellValue(3.14);
        assertEquals("3.14", ExcelExportUtils.getStringCellValue(decCell));

        // NUMERIC 日期格式
        Cell dateCell = row.createCell(3);
        dateCell.setCellValue(new java.util.Date());
        CellStyle dateStyle = wb.createCellStyle();
        dateStyle.setDataFormat((short) 14); // yyyy-mmdd
        dateCell.setCellStyle(dateStyle);
        String dateStr = ExcelExportUtils.getStringCellValue(dateCell);
        assertNotNull("日期字符串不应为 null", dateStr);
        assertFalse("日期字符串不应为空", dateStr.isEmpty());

        // BOOLEAN 类型
        Cell boolCell = row.createCell(4);
        boolCell.setCellValue(true);
        assertEquals("", ExcelExportUtils.getStringCellValue(boolCell)); // 非 STRING/NUMERIC 返回 ""

        // BLANK 类型
        Cell blankCell = row.createCell(5);
        blankCell.setBlank();
        assertEquals("", ExcelExportUtils.getStringCellValue(blankCell));

        try { wb.close(); } catch (Exception ignored) { }
    }
