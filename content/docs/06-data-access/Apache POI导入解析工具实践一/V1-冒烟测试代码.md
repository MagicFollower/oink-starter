package com.demo;

import org.apache.poi.hssf.usermodel.HSSFWorkbook;
import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.ss.usermodel.Sheet;
import org.apache.poi.ss.usermodel.Workbook;
import org.junit.Test;

import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.util.Arrays;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.*;

/**
 * ExcelImportParser 冒烟测试
 * 覆盖：基本解析、空行跳过、单元格补齐、表头校验、关闭校验、自定义起始行、titleIndexMap、异常包装
 */
public class ExcelImportParserSmokeTest {

    // ==================== 测试用 VO ====================

    static class TestVO {
        private String name;
        private String age;
        private String city;

        String getName() { return name; }
        void setName(String name) { this.name = name; }
        String getAge() { return age; }
        void setAge(String age) { this.age = age; }
        String getCity() { return city; }
        void setCity(String city) { this.city = city; }
    }

    // ==================== A1: 基本解析 ====================

    /**
     * 正常 xls 文件，验证表头校验 + 数据行映射 + getCellStr 工具方法
     */
    @Test
    public void testBasicParse() throws Exception {
        InputStream is = createXls(
                new String[]{"姓名", "年龄", "城市"},
                new String[][]{
                        {"张三", "25", "北京"},
                        {"李四", "30", "上海"}
                });

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄", "城市"))
                .mapper((rowIndex, cellList, titleMap) -> {
                    TestVO vo = new TestVO();
                    vo.setName(ExcelImportParser.getCellStr(cellList, titleMap, "姓名"));
                    vo.setAge(ExcelImportParser.getCellStr(cellList, titleMap, "年龄"));
                    vo.setCity(ExcelImportParser.getCellStr(cellList, titleMap, "城市"));
                    return vo;
                })
                .parse();

        assertEquals(2, result.size());
        assertEquals("张三", result.get(0).getName());
        assertEquals("25", result.get(0).getAge());
        assertEquals("北京", result.get(0).getCity());
        assertEquals("李四", result.get(1).getName());
        assertEquals("30", result.get(1).getAge());
        assertEquals("上海", result.get(1).getCity());
    }

    // ==================== A2: 空行跳过 ====================

    /**
     * 含空行的文件，验证空行被跳过，仅返回非空数据行
     */
    @Test
    public void testEmptyRowSkipped() throws Exception {
        Workbook wb = new HSSFWorkbook();
        Sheet sheet = wb.createSheet();
        // 表头
        Row h = sheet.createRow(0);
        h.createCell(0).setCellValue("姓名");
        h.createCell(1).setCellValue("年龄");
        // 数据行 1
        Row r1 = sheet.createRow(1);
        r1.createCell(0).setCellValue("张三");
        r1.createCell(1).setCellValue("25");
        // 空行（创建行但不设值，所有单元格为 blank/空串）
        sheet.createRow(2);
        // 数据行 2
        Row r3 = sheet.createRow(3);
        r3.createCell(0).setCellValue("李四");
        r3.createCell(1).setCellValue("30");

        InputStream is = toInputStream(wb);
        wb.close();

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄"))
                .mapper((rowIndex, cellList, titleMap) -> {
                    TestVO vo = new TestVO();
                    vo.setName(ExcelImportParser.getCellStr(cellList, titleMap, "姓名"));
                    vo.setAge(ExcelImportParser.getCellStr(cellList, titleMap, "年龄"));
                    return vo;
                })
                .parse();

        assertEquals("空行应被跳过，仅 2 条数据", 2, result.size());
        assertEquals("张三", result.get(0).getName());
        assertEquals("李四", result.get(1).getName());
    }

    // ==================== A3: 单元格补齐 ====================

    /**
     * 列数不足的行，验证补齐为空串
     */
    @Test
    public void testCellPadding() throws Exception {
        // 表头 3 列，数据行只有 1 列有值
        Workbook wb = new HSSFWorkbook();
        Sheet sheet = wb.createSheet();
        Row h = sheet.createRow(0);
        h.createCell(0).setCellValue("姓名");
        h.createCell(1).setCellValue("年龄");
        h.createCell(2).setCellValue("城市");

        Row r1 = sheet.createRow(1);
        r1.createCell(0).setCellValue("张三");
        // 年龄和城市列缺失

        InputStream is = toInputStream(wb);
        wb.close();

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄", "城市"))
                .mapper((rowIndex, cellList, titleMap) -> {
                    TestVO vo = new TestVO();
                    vo.setName(ExcelImportParser.getCellStr(cellList, titleMap, "姓名"));
                    vo.setAge(ExcelImportParser.getCellStr(cellList, titleMap, "年龄"));
                    vo.setCity(ExcelImportParser.getCellStr(cellList, titleMap, "城市"));
                    return vo;
                })
                .parse();

        assertEquals(1, result.size());
        assertEquals("张三", result.get(0).getName());
        assertEquals("缺失列应补齐为空串", "", result.get(0).getAge());
        assertEquals("缺失列应补齐为空串", "", result.get(0).getCity());
    }

    // ==================== A4: 表头不匹配 ====================

    /**

    // ==================== A5: 关闭表头校验 ====================

    /**
     * validateHeader(false) 时不校验表头，即使不匹配也不抛异常
     */
    @Test
    public void testSkipHeaderValidation() throws Exception {
        InputStream is = createXls(
                new String[]{"名字", "年纪"},   // 与 expected 不同
                new String[][]{{"张三", "25"}});

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄"))
                .validateHeader(false)   // 关闭校验
                .mapper((rowIndex, cellList, titleMap) -> {
                    TestVO vo = new TestVO();
                    vo.setName(cellList.get(0).toString().trim());
                    vo.setAge(cellList.get(1).toString().trim());
                    return vo;
                })
                .parse();

        assertEquals("关闭校验后应正常解析", 1, result.size());
        assertEquals("张三", result.get(0).getName());
    }

    // ==================== A6: 自定义数据起始行 ====================

    /**
     * dataStartRow(2) 跳过第 1 行（说明行），从第 2 行开始解析
     */
    @Test
    public void testCustomDataStartRow() throws Exception {
        Workbook wb = new HSSFWorkbook();
        Sheet sheet = wb.createSheet();
        // 第 0 行：表头
        Row h = sheet.createRow(0);
        h.createCell(0).setCellValue("姓名");
        h.createCell(1).setCellValue("年龄");
        // 第 1 行：说明行（应被跳过）
        Row desc = sheet.createRow(1);
        desc.createCell(0).setCellValue("请填写信息");
        desc.createCell(1).setCellValue("请勿留空");
        // 第 2 行：实际数据
        Row r = sheet.createRow(2);
        r.createCell(0).setCellValue("张三");
        r.createCell(1).setCellValue("25");

        InputStream is = toInputStream(wb);
        wb.close();

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄"))
                .dataStartRow(2)   // 从第 2 行开始解析
                .mapper((rowIndex, cellList, titleMap) -> {
                    TestVO vo = new TestVO();
                    vo.setName(ExcelImportParser.getCellStr(cellList, titleMap, "姓名"));
                    vo.setAge(ExcelImportParser.getCellStr(cellList, titleMap, "年龄"));
                    return vo;
                })
                .parse();

        assertEquals("应仅解析第 2 行数据", 1, result.size());
        assertEquals("张三", result.get(0).getName());
        assertEquals("25", result.get(0).getAge());
    }

    // ==================== A7: titleIndexMap 按列名取值 ====================

    /**
     * mapper 中通过列名取到正确的列索引，即使列顺序与期望不同
     */
    @Test
    public void testTitleIndexMap() throws Exception {
        // 注意：Excel 中列顺序为 "年龄", "姓名", "城市"（与 expectedTitles 顺序不同）
        InputStream is = createXls(
                new String[]{"年龄", "姓名", "城市"},
                new String[][]{{"25", "张三", "北京"}});

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("年龄", "姓名", "城市"))
                .mapper((rowIndex, cellList, titleMap) -> {
                    // 验证 titleIndexMap 内容正确
                    assertEquals("年龄列索引应为 0", Integer.valueOf(0), titleMap.get("年龄"));
                    assertEquals("姓名列索引应为 1", Integer.valueOf(1), titleMap.get("姓名"));
                    assertEquals("城市列索引应为 2", Integer.valueOf(2), titleMap.get("城市"));

                    TestVO vo = new TestVO();
                    // 按列名取值，不受列顺序影响
                    vo.setName(ExcelImportParser.getCellStr(cellList, titleMap, "姓名"));
                    vo.setAge(ExcelImportParser.getCellStr(cellList, titleMap, "年龄"));
                    vo.setCity(ExcelImportParser.getCellStr(cellList, titleMap, "城市"));
                    return vo;
                })
                .parse();

        assertEquals(1, result.size());
        assertEquals("按列名映射应正确", "张三", result.get(0).getName());
        assertEquals("按列名映射应正确", "25", result.get(0).getAge());
        assertEquals("按列名映射应正确", "北京", result.get(0).getCity());
    }

    /**
     * headerRowIndex(1) 指定表头在第 1 行，第 0 行为标题行被跳过，数据从第 2 行开始

        // 第 0 行：报表标题（应被跳过）
        Row title = sheet.createRow(0);

        // 第 1 行：表头（由 headerRowIndex 指定）

        h.createCell(0).setCellValue("姓名");
        h.createCell(1).setCellValue("年龄");
        // 第 2 行：数据
        Row r = sheet.createRow(2);
        r.createCell(0).setCellValue("张三");

        InputStream is = toInputStream(wb);

        List<TestVO> result = ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄"))
                .headerRowIndex(1)   // 表头在第 1 行
                .dataStartRow(2)     // 数据从第 2 行开始
                .mapper((rowIndex, cellList, titleMap) -> {

                    vo.setName(ExcelImportParser.getCellStr(cellList, titleMap, "姓名"));

    /**
     * headerRowIndex 指定错误行时，表头校验失败抛 BizException
     */
    @Test(expected = BizException.class)
    public void testHeaderRowIndex_Mismatch() throws Exception {
        Workbook wb = new HSSFWorkbook();
        Sheet sheet = wb.createSheet();
        // 第 0 行：标题行（不是表头）
        Row title = sheet.createRow(0);
        title.createCell(0).setCellValue("员工信息报表");
        // 第 1 行：真正的表头
        Row h = sheet.createRow(1);
        h.createCell(0).setCellValue("姓名");
        h.createCell(1).setCellValue("年龄");
        // 第 2 行：数据
        Row r = sheet.createRow(2);
        r.createCell(0).setCellValue("张三");
        r.createCell(1).setCellValue("25");

        InputStream is = toInputStream(wb);
        wb.close();

        // 错误地将 headerRowIndex 设为 0，第 0 行是标题不是表头，校验应失败
        ExcelImportParser.<TestVO>builder()
                .inputStream(is)
                .expectedTitles(Arrays.asList("姓名", "年龄"))
                .headerRowIndex(0)   // 指向了标题行，不匹配
                .dataStartRow(2)
                .mapper((rowIndex, cellList, titleMap) -> new TestVO())
                .parse();
    }

    // ==================== 辅助方法 ====================

    /**
     * 创建内存 xls 文件并转为 InputStream
     */
    private InputStream createXls(String[] titles, String[][] data) throws Exception {
        Workbook wb = new HSSFWorkbook();
        Sheet sheet = wb.createSheet();
        Row headerRow = sheet.createRow(0);
        for (int i = 0; i < titles.length; i++) {
            headerRow.createCell(i).setCellValue(titles[i]);
        }
        for (int r = 0; r < data.length; r++) {
            Row row = sheet.createRow(r + 1);
            for (int c = 0; c < data[r].length; c++) {
                row.createCell(c).setCellValue(data[r][c]);
            }
        }
        return toInputStream(wb);
    }

    /**
     * 将 Workbook 写为字节数组并转为 InputStream
     */
    private InputStream toInputStream(Workbook wb) throws Exception {
        ByteArrayOutputStream bos = new ByteArrayOutputStream();
        wb.write(bos);
        wb.close();
        return new ByteArrayInputStream(bos.toByteArray());
    }
}
