```java
package com.demo;

import cn.hutool.core.util.StrUtil;
import cn.hutool.poi.excel.ExcelUtil;
import cn.hutool.poi.excel.sax.handler.RowHandler;
import lombok.NonNull;

import java.io.InputStream;
import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * Excel 导入解析通用工具 V2
 * <pre>
 * 相比 V1 的改进：
 * 1. 引入 RowAccessor 封装行级数据访问，mapper 签名从 (rowIndex, cellList, titleMap) 简化为 (rowIndex, row)
 * 2. RowAccessor 支持多类型取值：getString / getInteger / getLong / getDecimal / getRaw
 * 3. 支持按列名取值（getString("列名")）和按索引取值（getStringAt(index)）
 * 4. 调用方代码更简洁，消除重复传递 cellList 和 titleMap 的冗余
 *
 * 使用示例：
 * <code>
 * List&lt;MyVO&gt; result = ExcelImportParserV2.&lt;MyVO&gt;builder()
 *     .inputStream(inputStream)
 *     .expectedTitles(Arrays.asList("姓名", "年龄", "金额"))
 *     .mapper((rowIndex, row) -&gt; {
 *         MyVO vo = new MyVO();
 *         vo.setName(row.getString("姓名"));
 *         vo.setAge(row.getInteger("年龄"));
 *         vo.setAmount(row.getDecimal("金额"));
 *         return vo;
 *     })
 *     .parse();
 * </code>
 * </pre>
 *
 * @param <T> 目标数据类型
 * @Version 2.0.0
 */
public class ExcelImportParserV2<T> {

    // ==================== 行数据访问器 ====================

    /**
     * 行数据访问器：封装 cellList + titleIndexMap，提供类型安全的取值方法
     * <p>
     * 每行数据解析时创建一个 RowAccessor 实例，调用方通过列名或索引获取单元格值。
     * 支持 String / Integer / Long / BigDecimal / Object 多种返回类型。
     * </p>
     */
    public static class RowAccessor {
        private final List<Object> cellList;
        private final Map<String, Integer> titleIndexMap;

        public RowAccessor(List<Object> cellList, Map<String, Integer> titleIndexMap) {
            this.cellList = cellList;
            this.titleIndexMap = titleIndexMap;
        }

        // ---------- 按列名取值 ----------

        /**
         * 按列名获取单元格的字符串值（trim 后），列不存在返回 ""
         */
        public String getString(String columnName) {
            Integer idx = titleIndexMap.get(columnName);
            if (idx == null || idx >= cellList.size()) {
                return "";
            }
            Object val = cellList.get(idx);
            return val != null ? val.toString().trim() : "";
        }

        /**
         * 按列名获取 Integer 值，列不存在或值为空返回 null
         * <p>自动处理 "25.0" 这种 Double toString 的情况</p>
         */
        public Integer getInteger(String columnName) {
            String s = getString(columnName);
            if (s.isEmpty()) {
                return null;
            }
            try {
                return new BigDecimal(s).intValue();
            } catch (NumberFormatException e) {
                return null;
            }
        }

        /**
         * 按列名获取 Long 值，列不存在或值为空返回 null
         */
        public Long getLong(String columnName) {
            String s = getString(columnName);
            if (s.isEmpty()) {
                return null;
            }
            try {
                return new BigDecimal(s).longValue();
            } catch (NumberFormatException e) {
                return null;
            }
        }

        /**
         * 按列名获取 BigDecimal 值，列不存在或值为空返回 null
         */
        public BigDecimal getDecimal(String columnName) {
            String s = getString(columnName);
            if (s.isEmpty()) {
                return null;
            }
            try {
                return new BigDecimal(s);
            } catch (NumberFormatException e) {
                return null;
            }
        }

        /**
         * 按列名获取原始 Object 值（不做任何转换），列不存在返回 null
         */
        public Object getRaw(String columnName) {
            Integer idx = titleIndexMap.get(columnName);
            if (idx == null || idx >= cellList.size()) {
                return null;
            }
            return cellList.get(idx);
        }

        // ---------- 按索引取值 ----------

        /**
         * 按列索引获取单元格的字符串值（trim 后），索引越界返回 ""
         */
        public String getStringAt(int index) {
            if (index < 0 || index >= cellList.size()) {
                return "";
            }
            Object val = cellList.get(index);
            return val != null ? val.toString().trim() : "";
        }

        /**
         * 按列索引获取 Integer 值，索引越界或值为空返回 null
         */
        public Integer getIntegerAt(int index) {
            String s = getStringAt(index);
            if (s.isEmpty()) {
                return null;
            }
            try {
                return new BigDecimal(s).intValue();
            } catch (NumberFormatException e) {
                return null;
            }
        }

        /**
         * 按列索引获取 Long 值，索引越界或值为空返回 null
         */
        public Long getLongAt(int index) {
            String s = getStringAt(index);
            if (s.isEmpty()) {
                return null;
            }
            try {
                return new BigDecimal(s).longValue();
            } catch (NumberFormatException e) {
                return null;
            }
        }

        /**
         * 按列索引获取 BigDecimal 值，索引越界或值为空返回 null
         */
        public BigDecimal getDecimalAt(int index) {
            String s = getStringAt(index);
            if (s.isEmpty()) {
                return null;
            }
            try {
                return new BigDecimal(s);
            } catch (NumberFormatException e) {
                return null;
            }
        }

        /**
         * 按列索引获取原始 Object 值，索引越界返回 null
         */
        public Object getRawAt(int index) {
            if (index < 0 || index >= cellList.size()) {
                return null;
            }
            return cellList.get(index);
        }

        /**
         * 获取当前行的单元格列表（只读访问）
         */
        public List<Object> getCellList() {
            return cellList;
        }

        /**
         * 获取列名→索引映射（只读访问）
         */
        public Map<String, Integer> getTitleIndexMap() {
            return titleIndexMap;
        }
    }

    // ==================== 函数式接口 ====================

    /**
     * 行数据映射器：将一行 Excel 数据转换为目标对象
     * <p>V2 简化为 (rowIndex, row) 两个参数，row 为 RowAccessor 实例</p>
     */
    @FunctionalInterface
    public interface RowMapper<T> {
        /**
         * @param rowIndex 行号（0-based）
         * @param row      行数据访问器，支持 getString/getInteger/getLong/getDecimal 等方法
         * @return 目标对象
         */
        T map(long rowIndex, RowAccessor row);
    }

    // ==================== 字段 ====================

    private InputStream inputStream;
    private List<String> expectedTitles;
    private int sheetIndex = 0;
    private boolean validateHeader = true;
    private long headerRowIndex = 0;
    private long dataStartRow = 1;
    private RowMapper<T> mapper;

    private ExcelImportParserV2() {
    }

    // ==================== 静态工厂 ====================

    /**
     * 创建 Builder 实例
     */
    public static <T> ExcelImportParserV2<T> builder() {
        return new ExcelImportParserV2<>();
    }

    // ==================== Builder 链式方法 ====================

    public ExcelImportParserV2<T> inputStream(@NonNull InputStream inputStream) {
        this.inputStream = inputStream;
        return this;
    }

    public ExcelImportParserV2<T> expectedTitles(@NonNull List<String> expectedTitles) {
        this.expectedTitles = new ArrayList<>(expectedTitles);
        return this;
    }

    public ExcelImportParserV2<T> sheetIndex(int sheetIndex) {
        this.sheetIndex = sheetIndex;
        return this;
    }

    public ExcelImportParserV2<T> validateHeader(boolean validateHeader) {
        this.validateHeader = validateHeader;
        return this;
    }

    /**
     * 指定表头行号（0-based），默认 0
     * <p>表头行之前的行会被跳过，表头行之后的行根据 dataStartRow 决定是否作为数据行解析。</p>
     */
    public ExcelImportParserV2<T> headerRowIndex(long headerRowIndex) {
        this.headerRowIndex = headerRowIndex;
        return this;
    }

    public ExcelImportParserV2<T> dataStartRow(long dataStartRow) {
        this.dataStartRow = dataStartRow;
        return this;
    }

    public ExcelImportParserV2<T> mapper(@NonNull RowMapper<T> mapper) {
        this.mapper = mapper;
        return this;
    }

    // ==================== 核心解析 ====================

    /**
     * 执行解析
     *
     * @return 解析结果列表
     * @throws BizException 表头校验失败或文件解析异常
     */
    public List<T> parse() {
        if (inputStream == null) {
            throw new IllegalArgumentException("inputStream 不能为空");
        }
        if (expectedTitles == null || expectedTitles.isEmpty()) {
            throw new IllegalArgumentException("expectedTitles 不能为空");
        }
        if (mapper == null) {
            throw new IllegalArgumentException("mapper 不能为空");
        }

        List<T> result = new ArrayList<>();
        Map<String, Integer> titleIndexMap = buildTitleIndexMap(expectedTitles);
        int columnCount = expectedTitles.size();

        try {
            ExcelUtil.readBySax(inputStream, sheetIndex, new RowHandler() {
                @Override
                public void handle(int sheetIndex, long rowIndex, List<Object> cellList) {
                    // 表头行：校验标题
                    if (rowIndex == headerRowIndex && validateHeader) {
                        validateHeaders(cellList);
                    }
                    // 数据行
                    if (rowIndex >= dataStartRow) {
                        if (isRowEmpty(cellList)) {
                            return;
                        }
                        padCellList(cellList, columnCount);
                        // V2 改进：创建 RowAccessor 封装行数据
                        RowAccessor row = new RowAccessor(cellList, titleIndexMap);
                        T obj = mapper.map(rowIndex, row);
                        if (obj != null) {
                            result.add(obj);
                        }
                    }
                }
            });
        } catch (BizException e) {
            throw e;
        } catch (Exception e) {
            throw new BizException(
                    SysRetCodeConstants.FAILED.getCode(),
                    "文件解析异常，请确保文件完整且未被加密: " + e.getMessage(),
                    e);
        }

        return result;
    }

    // ==================== 私有方法 ====================

    /**
     * 校验表头行是否与期望标题匹配
     */
    private void validateHeaders(List<Object> actualHeaders) {
        List<String> actual = new ArrayList<>();
        for (Object cell : actualHeaders) {
            actual.add(cell != null ? StrUtil.trim(cell.toString()) : "");
        }
        for (int i = 0; i < expectedTitles.size(); i++) {
            String expected = StrUtil.trim(expectedTitles.get(i));
            String act = i < actual.size() ? actual.get(i) : "";
            if (!StrUtil.equals(expected, act)) {
                throw new BizException(
                        SysRetCodeConstants.FAILED.getCode(),
                        String.format("文件解析异常:标题行不匹配(解析列[%s],excel中对应列为[%s])!", expected, act));
            }
        }
    }

    /**
     * 判断是否为空行（所有单元格为 null 或空白）
     */
    private boolean isRowEmpty(List<Object> cellList) {
        if (cellList == null || cellList.isEmpty()) {
            return true;
        }
        for (Object cell : cellList) {
            if (cell != null && StrUtil.isNotBlank(cell.toString())) {
                return false;
            }
        }
        return true;
    }

    /**
     * 补齐单元格列表：null 替换为空串，长度不足时用空串补齐
     */
    private void padCellList(List<Object> cellList, int expectedSize) {
        int currentSize = cellList.size();
        if (currentSize < expectedSize) {
            for (int i = 0; i < expectedSize - currentSize; i++) {
                cellList.add(StrUtil.EMPTY);
            }
        }
        for (int i = 0; i < cellList.size(); i++) {
            if (cellList.get(i) == null) {
                cellList.set(i, StrUtil.EMPTY);
            }
        }
    }

    /**
     * 构建列名 → 列索引映射
     */
    private Map<String, Integer> buildTitleIndexMap(List<String> titles) {
        Map<String, Integer> map = new HashMap<>();
        for (int i = 0; i < titles.size(); i++) {
            map.put(StrUtil.trim(titles.get(i)), i);
        }
        return map;
    }
}
```

## 二、V1 → V2 改进清单

| 编号 | V1 问题 | V2 改进 | 类别 |
|------|---------|---------|------|
| 1 | `getCellStr(cellList, titleMap, "列名")` 每次传 3 个参数 | `row.getString("列名")` 只传 1 个参数 | 参数冗余消除 |
| 2 | 只返回 String，数值需调用方手动转换 | `getInteger` / `getLong` / `getDecimal` 多类型支持 | 类型安全 |
| 3 | `cellList.get(0).toString().trim()` 硬编码索引 | `row.getStringAt(0)` 索引取值 + 边界保护 | 可读性 + 安全性 |
| 4 | 无法获取原始 Object 值 | `row.getRaw("列名")` / `row.getRawAt(index)` | 底层访问 |
| 5 | mapper 签名 `(rowIndex, cellList, titleMap)` 冗长 | mapper 签名 `(rowIndex, row)` 简洁 | 接口简化 |
| 6 | 漏传或错传参数时编译器不报错 | RowAccessor 方法调用，编译期可检查方法名 | 易错性降低 |

## 三、V2 新增核心组件：RowAccessor

### 设计思路

将 V1 中散落在 mapper 外部的 `cellList` 和 `titleIndexMap` 封装为一个行级访问器对象，对外暴露语义清晰的取值方法。

### 方法矩阵

| 维度 | 按列名 | 按索引 |
|------|--------|--------|
| String | `getString(columnName)` → 空返回 `""` | `getStringAt(index)` → 越界返回 `""` |
| Integer | `getInteger(columnName)` → 空返回 `null` | `getIntegerAt(index)` → 越界返回 `null` |
| Long | `getLong(columnName)` → 空返回 `null` | `getLongAt(index)` → 越界返回 `null` |
| BigDecimal | `getDecimal(columnName)` → 空返回 `null` | `getDecimalAt(index)` → 越界返回 `null` |
| Object | `getRaw(columnName)` → 不存在返回 `null` | `getRawAt(index)` → 越界返回 `null` |

### 数值转换策略

所有数值方法（`getInteger` / `getLong` / `getDecimal`）统一通过 `BigDecimal` 中转：

```
getString(columnName) → "25.0" → new BigDecimal("25.0") → intValue() → 25
```

**优势**：

- 自动处理 SAX 模式下数值单元格产生的 `"25.0"` 格式
- `BigDecimal` 中转确保精度不丢失
- 转换失败（非数字字符串）返回 `null`，不抛异常
