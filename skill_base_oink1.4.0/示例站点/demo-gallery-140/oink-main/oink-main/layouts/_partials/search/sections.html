{{- /* Read heading IDs from rendered HTML, including render hooks/shortcodes. Offsets are UTF-8 bytes in the emitted body, not guessed heading slugs or JavaScript character positions. Body text is stored once per page. */ -}} {{- $content := .page.Content | string -}} {{- $content = replaceRE `(?is)<(script|style)\b[^>]*>.*?` "" $content -}} {{- $opening := `(?is)"']|"[^"]*"|'[^']*')*>` -}} {{- $pattern := printf `%s.*?` $opening -}} {{- $marker := "\x1eOINK-SEARCH\x1e" -}} {{- $chunks := split (replaceRE $pattern (printf "%s${0}" $marker) $content) $marker -}} {{- $sections := slice -}} {{- $body := "" -}} {{- range $chunks -}} {{- $heading := "" -}} {{- range findRE (printf "^%s" $pattern) . 1 }}{{ $heading = . }}{{ end -}} {{- if $heading -}} {{- $id := "" -}} {{- range findRE $opening $heading 1 -}} {{- range findRE `\s+[^\s=/>]+(?:\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+))?` . -}} {{- if findRE `(?i)^\s+id\s*=` . -}} {{- $id = replaceRE `(?is)^\s+id\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+))$` "$1$2$3" . | htmlUnescape -}} {{- end -}} {{- end -}} {{- end -}} {{- $title := $heading | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- if $id -}} {{- $entry := dict "id" $id "title" $title -}} {{- if eq $.scope "content" }}{{ $entry = merge $entry (dict "start" (len $body)) }}{{ end -}} {{- $sections = $sections | append $entry -}} {{- end -}} {{- end -}} {{- if eq $.scope "content" -}} {{- $text := replaceRE `(?is)<(script|style)\b[^>]*>.*?` "" . -}} {{- $text = replaceRE `(?i)|` " " $text -}} {{- $text = $text | plainify | htmlUnescape -}} {{- $text = replaceRE `\s+` " " $text | strings.TrimSpace -}} {{- if $text }}{{ $body = printf "%s%s " $body $text }}{{ end -}} {{- end -}} {{- end -}} {{- return (dict "sections" $sections "body" $body) -}}
